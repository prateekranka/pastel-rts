function listFiles(root) {
  const output = [];
  const visit = (directory) => {
    for (const entry of readdirSync(directory, { withFileTypes: true })) {
      const absolute = join(directory, entry.name);
      if (entry.isDirectory()) {
        visit(absolute);
      } else if (entry.isFile()) {
        const bytes = readFileSync(absolute);
        output.push({ path: relative(root, absolute).split('\\').join('/'), byteLength: bytes.length, sha256: sha256(bytes) });
      }
    }
  };
  visit(root);
  output.sort((left, right) => left.path.localeCompare(right.path));
  return output;
}

async function requestJson(path, init = {}) {
  const response = await fetch(`${CONTENT_ORIGIN}${path}`, {
    ...init,
    headers: {
      ...(init.body === undefined ? {} : { 'content-type': 'application/json' }),
      ...(init.headers ?? {}),
    },
  });
  const text = await response.text();
  let body = null;
  try {
    body = text.length > 0 ? JSON.parse(text) : null;
  } catch {
    body = { raw: text.slice(0, 2000) };
  }
  if (!response.ok) {
    const detail = body && typeof body === 'object' && 'error' in body ? String(body.error) : text.slice(0, 300);
    throw new Error(`${init.method ?? 'GET'} ${path} returned HTTP ${String(response.status)}: ${detail}`);
  }
  return { status: response.status, body };
}

async function waitForAcknowledgement(revision, simulationRulesHash, timeoutMs = 30_000) {
  const deadline = Date.now() + timeoutMs;

async function waitForHttp(url, label, timeoutMs = 60_000) {
  const deadline = Date.now() + timeoutMs;
  let last = 'no response';
  while (Date.now() < deadline) {
    try {
      const response = await fetch(url, { signal: AbortSignal.timeout(1500) });
      if (response.ok) {
        console.log(`${label} ready: ${url}`);
        return;
      }
      last = `HTTP ${String(response.status)}`;
    } catch (error) {
      last = errorText(error);
    }
    await delay(100);
  }
  throw new Error(`${label} did not become ready at ${url}: ${last}`);
}

function processGroupAlive(pid) {
  if (!pid) {
    return false;
  }
  try {
    process.kill(-pid, 0);
    return true;
  } catch (error) {
    return error?.code !== 'ESRCH';
  }
}

async function stopOwnedStudio() {
  if (!service?.pid) {
    return;
  }
  stopping = true;
  const pid = service.pid;
  try {
    process.kill(-pid, 'SIGTERM');
  } catch (error) {
    if (error?.code !== 'ESRCH') {
      console.error(`SIGTERM failed for studio process group ${String(pid)}: ${errorText(error)}`);
    }
  }
  const deadline = Date.now() + 5_000;
  while (Date.now() < deadline && processGroupAlive(pid)) {
    await delay(50);
  }
  if (processGroupAlive(pid)) {
    console.error(`Studio process group ${String(pid)} remained after SIGTERM; sending SIGKILL.`);
    try {
      process.kill(-pid, 'SIGKILL');
    } catch (error) {
      if (error?.code !== 'ESRCH') {
        console.error(`SIGKILL failed for studio process group ${String(pid)}: ${errorText(error)}`);
      }
    }
    const killDeadline = Date.now() + 3_000;
    while (Date.now() < killDeadline && processGroupAlive(pid)) {
      await delay(50);
    }
  }
  if (serviceExit === null) {
    serviceExit = { code: service.exitCode, signal: service.signalCode, observedAfterStop: true };
  }
}

function attachBrowserEvents(currentPage) {
  currentPage.on('console', (message) => {
    browserEvents.console.push({
      type: message.type(),
      text: message.text(),
      location: message.location(),
      at: nowIso(),
    });
  });
  currentPage.on('pageerror', (error) => {
    browserEvents.pageErrors.push({ message: error.message, stack: error.stack ?? null, at: nowIso() });
  });
  currentPage.on('requestfailed', (request) => {
    browserEvents.requestFailed.push({
      method: request.method(),
      url: request.url(),
      failure: request.failure()?.errorText ?? null,
      at: nowIso(),
    });
  });
  currentPage.on('response', (response) => {
    if (response.status() >= 400) {
      browserEvents.httpErrors.push({ method: response.request().method(), url: response.url(), status: response.status(), at: nowIso() });
    }
  });
}

function isIncidentalBrowserError(entry) {
  return entry.kind === 'http' && entry.url.endsWith('/favicon.ico');
}

function currentBrowserErrorSummary() {
  const all = [
    ...browserEvents.console.map((entry) => ({ kind: 'console', ...entry })),
    ...browserEvents.pageErrors.map((entry) => ({ kind: 'pageerror', ...entry })),
    ...browserEvents.requestFailed.map((entry) => ({ kind: 'requestfailed', ...entry })),
    ...browserEvents.httpErrors.map((entry) => ({ kind: 'http', ...entry })),
  ];
  const runtimeErrors = all.filter((entry) => {
    if (entry.kind === 'console') {
      return entry.type === 'error';
    }
    if (entry.kind === 'http') {
      return !isIncidentalBrowserError(entry);
    }
    return true;
  });
  const incidental = all.filter(isIncidentalBrowserError);
  return {
    totalEvents: all.length,
    consoleMessages: browserEvents.console.length,
    consoleErrors: browserEvents.console.filter((entry) => entry.type === 'error').length,
    pageErrors: browserEvents.pageErrors.length,
    requestFailed: browserEvents.requestFailed.length,
    httpErrors: browserEvents.httpErrors.length,
    runtimeErrorCount: runtimeErrors.length,
    incidentalCount: incidental.length,
    runtimeErrors: runtimeErrors.slice(-100),
  };
}

async function startSampler() {
  await page.evaluate(() => {
    const previous = window.__m11qSampler;
    previous?.stop?.();
    const state = {
      startedAt: performance.now(),
      lastFrameAt: null,
      frames: [],
      metrics: [],
      nextMetricAt: 0,
      stopped: false,
      raf: 0,
      droppedFrames: 0,
    };
    const maxFrames = 200_000;
    const readMetric = (now) => {
      const app = window.__pastelApp;
      const lab = app?.getInteractionLab?.();
      const adapter = app?.getRenderer?.();
      let renderer = null;
      try {
        if (adapter) {
          const stats = adapter.getStats();
          const resources = adapter.getResourceCounts();
          renderer = {
            kind: adapter.kind,
            requested: adapter.requested,
            backend: adapter.backend,
            initError: adapter.initError,
            drawingBuffer: adapter.getDrawingBufferSize(),
            pixelRatio: adapter.getPixelRatio(),
            stats: { drawCalls: stats.drawCalls, triangles: stats.triangles },
            resources: { textures: resources.textures, geometries: resources.geometries },
          };
        }
      } catch (error) {
        renderer = { error: String(error) };
      }
      const memory = performance.memory
        ? {
            available: true,
            usedJSHeapSize: performance.memory.usedJSHeapSize,
            totalJSHeapSize: performance.memory.totalJSHeapSize,
            jsHeapSizeLimit: performance.memory.jsHeapSizeLimit,
          }
        : { available: false, reason: 'not exposed' };
      return {
        atMs: now - state.startedAt,
        runtime: lab?.runtime?.getDiagnostics?.() ?? null,
        diagnostics: lab?.getDiagnostics?.() ?? null,
        status: app?.getContentStatus?.() ?? null,
        renderer,
        heap: memory,
      };
    };
    const loop = (now) => {
      const atMs = now - state.startedAt;
      if (state.lastFrameAt !== null) {
        if (state.frames.length < maxFrames) {
          state.frames.push([atMs, now - state.lastFrameAt]);
        } else {
          state.droppedFrames += 1;
        }
      }
      state.lastFrameAt = now;
      if (atMs >= state.nextMetricAt) {
        state.metrics.push(readMetric(now));
        state.nextMetricAt += 1000;
      }
      if (!state.stopped) {
        state.raf = requestAnimationFrame(loop);
      }
    };
    state.raf = requestAnimationFrame(loop);
    window.__m11qSampler = {
      latest: () => state.metrics[state.metrics.length - 1] ?? null,
      stop: () => {
        if (!state.stopped) {
          state.stopped = true;
          cancelAnimationFrame(state.raf);
        }
        return {
          startedAtBrowserMs: state.startedAt,
          stoppedAtBrowserMs: performance.now(),
          frames: state.frames,
          metrics: state.metrics,
          droppedFrames: state.droppedFrames,
        };
      },
    };
  });
}

function percentile(sorted, p) {
  if (sorted.length === 0) {
    return 0;
  }
  const index = Math.min(sorted.length - 1, Math.max(0, Math.ceil(p * sorted.length) - 1));
  return sorted[index] ?? 0;
}

function average(values) {
  if (values.length === 0) {
    return 0;
  }
  return values.reduce((total, value) => total + value, 0) / values.length;
}

function summarizeFrames(framePairs, fromAtMs = 0, toAtMs = Number.POSITIVE_INFINITY) {
  const values = framePairs
    .filter((pair) => pair[0] >= fromAtMs && pair[0] < toAtMs)
    .map((pair) => pair[1])
    .filter((value) => Number.isFinite(value) && value > 0);
  const sorted = [...values].sort((left, right) => left - right);
  const avgMs = average(values);
  const p99Ms = percentile(sorted, 0.99);
  return {
    sampleCount: values.length,
    minFrameMs: sorted[0] ?? 0,
    avgFrameMs: avgMs,
    p50FrameMs: percentile(sorted, 0.5),
    p95FrameMs: percentile(sorted, 0.95),
    p99FrameMs: p99Ms,
    maxFrameMs: sorted[sorted.length - 1] ?? 0,
    avgFps: avgMs > 0 ? 1000 / avgMs : 0,
    onePercentLowFps: p99Ms > 0 ? 1000 / p99Ms : 0,
    longFrameCountAtOrAbove50ms: values.filter((value) => value >= 50).length,
    source: 'requestAnimationFrame interval on composited browser page; not GPU time',
  };
}

function seriesValues(metrics, selector, fromAtMs = 0) {
  return metrics
    .filter((metric) => metric.atMs >= fromAtMs)
    .map(selector)
    .filter((value) => Number.isFinite(value));
}

function summarizeNumberSeries(values) {
  const sorted = [...values].sort((left, right) => left - right);
  return {
    sampleCount: values.length,
    min: sorted[0] ?? 0,
    avg: average(values),
    p50: percentile(sorted, 0.5),
    p95: percentile(sorted, 0.95),
    p99: percentile(sorted, 0.99),
    max: sorted[sorted.length - 1] ?? 0,
  };
}

function summarizeMetricSeries(metrics, fromAtMs) {
  const sim = summarizeNumberSeries(seriesValues(metrics, (metric) => metric.runtime?.tickDurationMs, fromAtMs));
  const nav = summarizeNumberSeries(seriesValues(metrics, (metric) => metric.runtime?.navDurationMs, fromAtMs));
  const latency = summarizeNumberSeries(seriesValues(metrics, (metric) => metric.runtime?.snapshotLatencyMs, fromAtMs));
  const drawCalls = summarizeNumberSeries(seriesValues(metrics, (metric) => metric.renderer?.stats?.drawCalls, fromAtMs));
  const triangles = summarizeNumberSeries(seriesValues(metrics, (metric) => metric.renderer?.stats?.triangles, fromAtMs));
  return {
    simulationTickDurationMs: sim,
    navigationDurationMs: nav,
    snapshotLatencyMs: latency,
    drawCalls,
    triangles,
  };
}

function summarizeResourceSeries(metrics, fromAtMs = 0) {
  const relevant = metrics.filter((metric) => metric.atMs >= fromAtMs && metric.renderer?.resources);
  const textures = relevant.map((metric) => metric.renderer.resources.textures).filter(Number.isFinite);
  const geometries = relevant.map((metric) => metric.renderer.resources.geometries).filter(Number.isFinite);
  const drawCalls = relevant.map((metric) => metric.renderer?.stats?.drawCalls).filter(Number.isFinite);
  const unique = (values) => [...new Set(values)].sort((left, right) => left - right);
  return {
    sampleCount: relevant.length,
    textures: { min: Math.min(...textures, 0), max: Math.max(...textures, 0), unique: unique(textures) },
    geometries: { min: Math.min(...geometries, 0), max: Math.max(...geometries, 0), unique: unique(geometries) },
    drawCalls: { min: Math.min(...drawCalls, 0), max: Math.max(...drawCalls, 0), unique: unique(drawCalls) },
  };
}

async function captureCheckpoint(label, phase, contextData = {}) {
  if (!page) {
    throw new Error(`Cannot capture ${label}: browser page is unavailable`);
  }
  const screenshotPath = join(ARTIFACT_DIR, `${label}.png`);
  const screenshot = await page.screenshot({ path: screenshotPath });
  const screenshotValid = screenshot.length >= 1000 && screenshot[0] === 0x89 && screenshot[1] === 0x50 && screenshot.readUInt32BE(16) === 1280 && screenshot.readUInt32BE(20) === 800;
  const runtime = await readPageSnapshot();
  const heap = await readHeapUsage();
  const record = {
    schemaVersion: 1,
    evidence: 'M1.1-Q',
    label,
    phase,
    capturedAt: nowIso(),
    wallElapsedMs: Date.now() - startedAtWall,
    soakElapsedMs: soakStartedAtWall === null ? null : Date.now() - soakStartedAtWall,
    screenshotPath,
    screenshotBytes: screenshot.length,
    screenshotDimensions: { width: screenshot.readUInt32BE(16), height: screenshot.readUInt32BE(20) },
    screenshotValid,
    runtime,
    heap,
    browserErrors: currentBrowserErrorSummary(),
    ...contextData,
  };
  writeJson(join(ARTIFACT_DIR, `${label}.json`), record);
  checkpoints.push({ label, phase, path: join(ARTIFACT_DIR, `${label}.json`), screenshotPath, capturedAt: record.capturedAt });
  checkObserved(screenshotValid, `Screenshot ${label} is not a valid composited 1280x800 PNG`, { path: screenshotPath });
  return record;
}

  alternatePngBase64,

  alternateSha,
  baselineSimulationRulesHash,
  previousTick,
  previousVisualContentHash,
) {
  const blockerCountBefore = blockers.length;
  const target = cycle % 2 === 1 ? 'alternate-gravemark-stalker' : 'original-sunweaver-scout';
  const expectedInputSha256 = target === 'alternate-gravemark-stalker' ? alternateSha : originalSha;
  const pngBase64 = target === 'alternate-gravemark-stalker' ? alternatePngBase64 : originalPngBase64;
  const record = {
    cycle,
    target,
    expectedInputSha256,
    baselineSimulationRulesHash,
    previousTick,
    previousVisualContentHash,
    startedAt: nowIso(),
    scheduledSoakElapsedMs: cycle * REPLACEMENT_INTERVAL_MS,
  };
  try {
    const draftBefore = await readDraftPack();
    const beforeScout = cloneJson(findScout(draftBefore));
    const archetype = cloneJson(beforeScout);
    archetype.displayName = `Sunweaver Scout Q Cycle ${String(cycle)}`;
    const mutationResponse = await requestJson(`/v2/units/${encodeURIComponent(SCOUT_ID)}`, {
      method: 'PUT',
      body: JSON.stringify({
        archetype,
        expectedDraftRevision: draftBefore.revision,
        pngBase64,
      }),
    });
    const mutation = mutationResponse.body;
    const validationResponse = await requestJson('/v2/validate', {
      method: 'POST',
      body: JSON.stringify({ expectedDraftRevision: mutation.draft.revision }),
    });
    const publicationBefore = await readPublication();
    const publishResponse = await requestJson('/v2/publish', {
      method: 'POST',
      body: JSON.stringify({ expectedRevision: publicationBefore.currentRevision }),
    });
    const published = publishResponse.body;
    const metadata = published.metadata;
    record.draftBeforeRevision = draftBefore.revision;
    record.draftAfterRevision = mutation.draft.revision;
    record.validation = { status: validationResponse.status, ok: validationResponse.body.ok, draftRevision: validationResponse.body.draftRevision };
    record.expectedPublishRevision = publicationBefore.currentRevision;
    record.revision = published.revision;
    record.metadata = metadata;
    record.publishStatus = publishResponse.status;
    record.revisionPack = await readRevisionPack(published.revision);
    const servedAssetResponse = await fetch(`${CONTENT_ORIGIN}/v2/revisions/${encodeURIComponent(published.revision)}/assets/${encodeAssetPath(SCOUT_ASSET_PATH)}`);
    const servedBytes = Buffer.from(await servedAssetResponse.arrayBuffer());
    const servedSha256 = sha256(servedBytes);
    record.asset = {
      path: SCOUT_ASSET_PATH,
      responseStatus: servedAssetResponse.status,
      byteLength: servedBytes.length,
      servedSha256,
      metadataSha256: findAsset(metadata, SCOUT_ASSET_PATH).sha256,
      expectedInputSha256,
      servedMatchesMetadata: servedSha256 === findAsset(metadata, SCOUT_ASSET_PATH).sha256,
      servedMatchesExpectedInput: servedSha256 === expectedInputSha256,
    };
    checkObserved(servedAssetResponse.ok, `Replacement ${String(cycle)} asset request failed`, { cycle, revision: published.revision });
    checkObserved(record.asset.servedMatchesMetadata, `Replacement ${String(cycle)} served asset does not match metadata`, { cycle, revision: published.revision, asset: record.asset });
    checkObserved(record.asset.servedMatchesExpectedInput, `Replacement ${String(cycle)} served asset does not match the requested source PNG`, { cycle, revision: published.revision, asset: record.asset });

    await waitForRevision(published.revision);
    const ack = await waitForAcknowledgement(published.revision, metadata.simulationRulesHash);
    await waitForArtReady();
    await delay(500);
    const runtime = await readPageSnapshot();
    record.acknowledgement = ack;
    record.runtime = runtime;
    record.finishedAt = nowIso();
    record.actualSoakElapsedMs = soakStartedAtWall === null ? null : Date.now() - soakStartedAtWall;
    checkObserved(runtime.content?.identity?.revision === published.revision, `Runtime did not adopt exact revision ${published.revision}`, { cycle, runtime: runtime.content, status: runtime.status });
    checkObserved(runtime.status?.activeRevision === published.revision && runtime.status?.phase === 'ready', `Runtime status is not ready on exact revision ${published.revision}`, { cycle, status: runtime.status });
    checkObserved(runtime.status?.pendingRevision === null, `Runtime retained a pending revision after cosmetic replacement ${published.revision}`, { cycle, status: runtime.status });
    checkObserved(runtime.diagnostics?.scenarioId === SCENARIO_ID && runtime.diagnostics?.seed === SEED, `Runtime scenario identity changed during replacement ${published.revision}`, { cycle, diagnostics: runtime.diagnostics });
    checkObserved(runtime.diagnostics?.entityCount === EXPECTED_ENTITY_COUNT, `Entity count changed during replacement ${published.revision}`, { cycle, diagnostics: runtime.diagnostics });
    checkObserved(metadata.restartRequired === false, `Cosmetic replacement ${published.revision} unexpectedly requires restart`, { cycle, metadata });
    checkObserved(metadata.simulationRulesHash === record.baselineSimulationRulesHash, `Simulation-rules hash changed during cosmetic replacement ${published.revision}`, { cycle, expected: record.baselineSimulationRulesHash, actual: metadata.simulationRulesHash });
    checkObserved(runtime.content?.identity?.simulationRulesHash === record.baselineSimulationRulesHash, `Runtime simulation-rules hash changed during cosmetic replacement ${published.revision}`, { cycle, runtime: runtime.content?.identity });
    checkObserved(runtime.diagnostics?.tick > record.previousTick, `Simulation tick did not advance across replacement ${published.revision}`, { cycle, previousTick: record.previousTick, runtimeTick: runtime.diagnostics?.tick });
    checkObserved(runtime.content?.assetBaseUrl === `${'/dev-content/v2/revisions/'}${published.revision}/assets/`, `Runtime asset base does not identify exact revision ${published.revision}`, { cycle, assetBaseUrl: runtime.content?.assetBaseUrl });
    record.ok = blockers.length === blockerCountBefore;
    if (!record.ok) {
      record.error = 'One or more replacement acceptance observations failed';
    }
  } catch (error) {
    record.ok = false;
    record.error = errorText(error);
    record.finishedAt = nowIso();
    record.actualSoakElapsedMs = soakStartedAtWall === null ? null : Date.now() - soakStartedAtWall;
    recordBlocker(`Replacement cycle ${String(cycle)} failed`, { cycle, error: record.error });
  }
  replacements.push(record);
  writeJson(join(ARTIFACT_DIR, `replacement-${String(cycle).padStart(2, '0')}.json`), record);
  return record;
}

function latestByAtMs(values, fromAtMs = 0) {
  const eligible = values.filter((value) => value.atMs >= fromAtMs);
  return eligible[eligible.length - 1] ?? null;
}

function countBy(values, selector) {
  const counts = new Map();
  for (const value of values) {
    const key = selector(value);
    counts.set(key, (counts.get(key) ?? 0) + 1);
  }
  return [...counts.entries()].sort((left, right) => right[1] - left[1]).map(([value, count]) => ({ value, count }));
}
